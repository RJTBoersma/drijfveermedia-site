# Drijfveer Media, Apache-instellingen (Antagonist). Gegenereerd door scripts/redirects.py; pas scripts/htaccess-basis.txt aan, niet dit bestand.
Options -Indexes
ErrorDocument 404 /404.html
DirectoryIndex index.html

RewriteEngine On
# Korte link voor klanten om een Google-review te schrijven: drijfveermedia.nl/review
RewriteRule ^review/?$ https://search.google.com/local/writereview?placeid=ChIJ3WI9wErNyUcR-qmlID9Y-YA [R=302,L,NE]
# @@OUDE-URLS@@
# /pad/index.html is dezelfde pagina als /pad/
RewriteCond %{THE_REQUEST} \s/+(.*/)?index\.html[\s?] [NC]
RewriteRule ^ https://www.drijfveermedia.nl/%1 [R=301,L]
# Altijd https en www, in één stap
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^(.*)$ https://www.drijfveermedia.nl/$1 [L,R=301]
# Mappen altijd met slash aan het eind
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_URI} !(\.[a-z0-9]{2,5}|/)$ [NC]
RewriteRule ^(.*)$ https://www.drijfveermedia.nl/$1/ [L,R=301]

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Strict-Transport-Security "max-age=31536000"
  # Bestanden met een hash in de naam (/_astro/) veranderen nooit; de rest een week bewaren
  <FilesMatch "\.(css|js|woff2|svg|webp|png|jpg|ico)$">
    Header set Cache-Control "public, max-age=604800"
  </FilesMatch>
  <If "%{REQUEST_URI} =~ m#^/_astro/#">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </If>
  <FilesMatch "\.html$">
    Header set Cache-Control "public, max-age=600"
  </FilesMatch>
</IfModule>
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml application/json text/plain application/xml
</IfModule>
